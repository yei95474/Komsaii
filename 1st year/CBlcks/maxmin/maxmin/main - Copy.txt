/*Prob: You are given an array of integers with a length of 𝑛. Your task is to determine and
output both the minimum and maximum values present in the array.*/

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int main()
{
    int array1[5] = {}, max=0, min=0;

    printf("Select 5 integers\n");

    for (int i=0; i<5; i++) {
        scanf("%d", &array1[i]);
    }

    for (int i=0; i<5; i++ ) {
        printf("%d ", array1[i]);
    }

    printf("\n");
    for (int i=0; i<5; i++ ) {
        if (array1[i]>max) {
            max=array1[i];
        }
        if (array1[i]<max) {
            min=array1[i];
        }
    }

    printf("Max=%d\n", max);
    printf("Min=%d\n", min);

return 0;
}
